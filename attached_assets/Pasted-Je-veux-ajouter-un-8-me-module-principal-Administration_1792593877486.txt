AXIABAT est une application React avec Firebase Authentication et Firestore.
Le module Administration est réservé aux utilisateurs ayant le rôle "administrateur" 
dans Firestore (collection "utilisateurs", champ "role" = "administrateur").

La sécurité doit être assurée côté serveur via les Firebase Security Rules,
pas uniquement côté interface.

═══════════════════════════════════════════
1. ARCHITECTURE DES RÔLES
═══════════════════════════════════════════

Les rôles disponibles dans Firestore sont :

- administrateur   → accès à tous les modules + Administration
- chef_projet      → Projets, Journal, Délais, Budget, Dashboard, Reporting
- ingenieur        → Projets, Journal, Délais, Dashboard
- conducteur       → Journal, Délais, Budget
- lecteur          → Dashboard, Reporting (lecture seule)

Chaque utilisateur est stocké dans Firestore :
Collection : utilisateurs
Document ID : uid Firebase Auth
Champs : nom, email, role, actif (boolean), dateExpiration (timestamp), modules (array), createdAt, createdBy

2. FIREBASE SECURITY RULES
═══════════════════════════════════════════

Mettre à jour firestore.rules avec ces règles :

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isAdmin() {
      return request.auth != null &&
        get(/databases/$(database)/documents/utilisateurs/$(request.auth.uid)).data.role == 'administrateur';
    }

    function isActive() {
      return request.auth != null &&
        get(/databases/$(database)/documents/utilisateurs/$(request.auth.uid)).data.actif == true;
    }

    // Utilisateurs : seul l'admin peut lire/écrire la liste complète
    match /utilisateurs/{uid} {
      allow read: if request.auth.uid == uid || isAdmin();
      allow create: if isAdmin();
      allow update: if isAdmin();
      allow delete: if isAdmin();
    }

    // Journal d'administration : lecture et écriture admin uniquement
    match /admin_logs/{logId} {
      allow read, write: if isAdmin();
    }

    // Projets : utilisateurs actifs seulement
    match /projets/{projetId} {
      allow read: if isActive();
      allow write: if isActive() && (
        get(/databases/$(database)/documents/utilisateurs/$(request.auth.uid)).data.role in 
        ['administrateur', 'chef_projet', 'ingenieur']
      );
      match /ArticlesProjet/{articleId} {
        allow read, write: if isActive();
      }
    }

    // Articles et catégories
    match /articles/{id} {
      allow read: if isActive();
      allow write: if isAdmin();
    }
    match /categories/{id} {
      allow read: if isActive();
      allow write: if isAdmin();
    }
  }
}

import { db, firebaseAuth } from '../firebase';
import {
  collection, doc, getDocs, getDoc, addDoc,
  updateDoc, deleteDoc, query, orderBy, serverTimestamp
} from 'firebase/firestore';
import { sendPasswordResetEmail } from 'firebase/auth';

const COLLECTION_USERS = 'utilisateurs';
const COLLECTION_LOGS  = 'admin_logs';

// Lister tous les utilisateurs
export async function listerUtilisateurs() {
  const snap = await getDocs(collection(db, COLLECTION_USERS));
  return snap.docs.map(d => ({ id: d.id, ...d.data() }));
}

// Ajouter un utilisateur dans Firestore (le compte Firebase Auth est créé séparément)
export async function ajouterUtilisateur(adminUid, data) {
  const ref = await addDoc(collection(db, COLLECTION_USERS), {
    nom: data.nom,
    email: data.email,
    role: data.role,
    actif: true,
    modules: data.modules || [],
    dateExpiration: data.dateExpiration || null,
    createdAt: serverTimestamp(),
    createdBy: adminUid,
  });
  await loggerAction(adminUid, 'CREATE_USER', { email: data.email, role: data.role });
  return ref.id;
}

// Modifier un utilisateur
export async function modifierUtilisateur(adminUid, uid, data) {
  await updateDoc(doc(db, COLLECTION_USERS, uid), {
    ...data,
    updatedAt: serverTimestamp(),
    updatedBy: adminUid,
  });
  await loggerAction(adminUid, 'UPDATE_USER', { uid, changes: Object.keys(data) });
}

// Activer / désactiver un compte
export async function toggleActif(adminUid, uid, actif) {
  await updateDoc(doc(db, COLLECTION_USERS, uid), {
    actif,
    updatedAt: serverTimestamp(),
    updatedBy: adminUid,
  });
  await loggerAction(adminUid, actif ? 'ACTIVATE_USER' : 'DEACTIVATE_USER', { uid });
}

// Changer le rôle
export async function changerRole(adminUid, uid, nouveauRole) {
  await updateDoc(doc(db, COLLECTION_USERS, uid), {
    role: nouveauRole,
    updatedAt: serverTimestamp(),
    updatedBy: adminUid,
  });
  await loggerAction(adminUid, 'CHANGE_ROLE', { uid, nouveauRole });
}

// Envoyer email de réinitialisation
export async function reinitialiserMotDePasse(adminUid, email) {
  await sendPasswordResetEmail(firebaseAuth, email);
  await loggerAction(adminUid, 'RESET_PASSWORD', { email });
}

// Supprimer un utilisateur de Firestore
export async function supprimerUtilisateur(adminUid, uid, email) {
  await deleteDoc(doc(db, COLLECTION_USERS, uid));
  await loggerAction(adminUid, 'DELETE_USER', { uid, email });
}

// Lire le journal d'administration
export async function lireLogs() {
  const q = query(collection(db, COLLECTION_LOGS), orderBy('timestamp', 'desc'));
  const snap = await getDocs(q);
  return snap.docs.map(d => ({ id: d.id, ...d.data() }));
}

// Enregistrer une action dans le journal
async function loggerAction(adminUid, action, details) {
  await addDoc(collection(db, COLLECTION_LOGS), {
    adminUid,
    action,
    details,
    timestamp: serverTimestamp(),
    resultat: 'SUCCESS',
  });
}

import React, { useState, useEffect } from 'react';
import {
  listerUtilisateurs, toggleActif,
  reinitialiserMotDePasse, supprimerUtilisateur
} from '../../../services/adminService';

const ROLES_LABELS = {
  administrateur: { label: 'Administrateur', color: '#7E57C2' },
  chef_projet:    { label: 'Chef de projet',  color: '#1565C0' },
  ingenieur:      { label: 'Ingénieur',        color: '#00897B' },
  conducteur:     { label: 'Conducteur',       color: '#F57F17' },
  lecteur:        { label: 'Lecteur',           color: '#546E7A' },
};

export default function ListeUtilisateurs({ adminUid, onEditer }) {
  const [utilisateurs, setUtilisateurs] = useState([]);
  const [chargement, setChargement]     = useState(true);
  const [message, setMessage]           = useState('');

  const charger = async () => {
    setChargement(true);
    try {
      const data = await listerUtilisateurs();
      setUtilisateurs(data);
    } catch(e) {
      setMessage('Erreur chargement : ' + e.message);
    } finally {
      setChargement(false);
    }
  };

  useEffect(() => { charger(); }, []);

  const handleToggle = async (uid, actif) => {
    try {
      await toggleActif(adminUid, uid, !actif);
      setMessage(!actif ? '✅ Compte activé' : '⚠️ Compte désactivé');
      charger();
    } catch(e) { setMessage('Erreur : ' + e.message); }
  };

  const handleReset = async (email) => {
    if (!window.confirm(`Envoyer un email de réinitialisation à ${email} ?`)) return;
    try {
      await reinitialiserMotDePasse(adminUid, email);
      setMessage('✅ Email de réinitialisation envoyé à ' + email);
    } catch(e) { setMessage('Erreur : ' + e.message); }
  };

  const handleSupprimer = async (uid, email) => {
    if (!window.confirm(`Supprimer définitivement le compte ${email} ?`)) return;
    try {
      await supprimerUtilisateur(adminUid, uid, email);
      setMessage('✅ Compte supprimé');
      charger();
    } catch(e) { setMessage('Erreur : ' + e.message); }
  };

  if (chargement) return <div className="admin-loading">Chargement...</div>;

  return (
    <div>
      {message && (
        <div className="admin-message" onClick={() => setMessage('')}>
          {message} ✕
        </div>
      )}
      <table className="admin-table">
        <thead>
          <tr>
            <th>Nom</th>
            <th>Email</th>
            <th>Rôle</th>
            <th>Statut</th>
            <th>Expiration</th>
            <th>Actions</th>
          </tr>
        </thead>
        <tbody>
          {utilisateurs.map(u => {
            const roleInfo = ROLES_LABELS[u.role] || { label: u.role, color: '#999' };
            return (
              <tr key={u.id}>
                <td>{u.nom || '—'}</td>
                <td>{u.email}</td>
                <td>
                  <span className="admin-role-badge"
                    style={{ background: roleInfo.color + '22', color: roleInfo.color,
                      padding:'2px 10px', borderRadius:20, fontSize:12, fontWeight:600 }}>
                    {roleInfo.label}
                  </span>
                </td>
                <td>
                  <span style={{ color: u.actif ? '#43A047' : '#E53935', fontWeight:600, fontSize:12 }}>
                    {u.actif ? '● Actif' : '● Suspendu'}
                  </span>
                </td>
                <td style={{ fontSize:12, color:'#78909C' }}>
                  {u.dateExpiration
                    ? new Date(u.dateExpiration.seconds * 1000).toLocaleDateString('fr-FR')
                    : '—'}
                </td>
                <td>
                  <div style={{ display:'flex', gap:6, flexWrap:'wrap' }}>
                    <button className="admin-btn-sm admin-btn-edit"
                      onClick={() => onEditer(u)}>✏️ Modifier</button>
                    <button className="admin-btn-sm"
                      style={{ background: u.actif ? '#FFF3E0' : '#E8F5E9',
                        color: u.actif ? '#E65100' : '#2E7D32' }}
                      onClick={() => handleToggle(u.id, u.actif)}>
                      {u.actif ? '⏸ Suspendre' : '▶ Activer'}
                    </button>
                    <button className="admin-btn-sm admin-btn-reset"
                      onClick={() => handleReset(u.email)}>🔑 Reset MDP</button>
                    <button className="admin-btn-sm admin-btn-delete"
                      onClick={() => handleSupprimer(u.id, u.email)}>🗑 Supprimer</button>
                  </div>
                </td>
              </tr>
            );
          })}
        </tbody>
      </table>
    </div>
  );
}

import React, { useState } from 'react';
import { ajouterUtilisateur, modifierUtilisateur, changerRole } from '../../../services/adminService';

const ROLES = ['administrateur','chef_projet','ingenieur','conducteur','lecteur'];
const MODULES_DISPO = ['projets','journal','delais','budget','dashboard','reporting','base_donnees'];

export default function FormulaireUtilisateur({ adminUid, userAEditer, onClose }) {
  const [form, setForm] = useState({
    nom:            userAEditer?.nom    || '',
    email:          userAEditer?.email  || '',
    role:           userAEditer?.role   || 'conducteur',
    modules:        userAEditer?.modules || MODULES_DISPO,
    dateExpiration: '',
  });
  const [erreur,      setErreur]      = useState('');
  const [chargement,  setChargement]  = useState(false);
  const [succes,      setSucces]      = useState('');

  const handleChange = (e) => {
    const { name, value } = e.target;
    setForm(p => ({ ...p, [name]: value }));
  };

  const toggleModule = (mod) => {
    setForm(p => ({
      ...p,
      modules: p.modules.includes(mod)
        ? p.modules.filter(m => m !== mod)
        : [...p.modules, mod],
    }));
  };

  const handleSave = async () => {
    if (!form.nom.trim() || !form.email.trim()) {
      setErreur('Nom et email sont obligatoires.'); return;
    }
    setChargement(true); setErreur('');
    try {
      if (userAEditer) {
        await modifierUtilisateur(adminUid, userAEditer.id, {
          nom: form.nom, modules: form.modules,
          dateExpiration: form.dateExpiration
            ? new Date(form.dateExpiration) : null,
        });
        if (form.role !== userAEditer.role) {
          await changerRole(adminUid, userAEditer.id, form.role);
        }
        setSucces('✅ Utilisateur modifié avec succès !');
      } else {
        await ajouterUtilisateur(adminUid, form);
        setSucces('✅ Utilisateur ajouté. Il doit maintenant créer son compte Firebase Auth avec cet email.');
      }
      setTimeout(onClose, 2000);
    } catch(e) {
      setErreur('Erreur : ' + e.message);
    } finally { setChargement(false); }
  };

  return (
    <div className="admin-modal-overlay">
      <div className="admin-modal">
        <div className="admin-modal-header">
          <h3>{userAEditer ? '✏️ Modifier l\'utilisateur' : '➕ Ajouter un utilisateur'}</h3>
          <button onClick={onClose} className="admin-modal-close">×</button>
        </div>

        <div className="admin-modal-body">
          {erreur  && <div className="admin-error">{erreur}</div>}
          {succes  && <div className="admin-succes">{succes}</div>}

          <div className="admin-field">
            <label>Nom complet *</label>
            <input name="nom" value={form.nom}
              onChange={handleChange} placeholder="Ex: Mohamed Amara" />
          </div>
          <div className="admin-field">
            <label>Email *</label>
            <input name="email" type="email" value={form.email}
              onChange={handleChange} placeholder="utilisateur@axiabat.com"
              disabled={!!userAEditer} />
            {userAEditer && <small>L'email ne peut pas être modifié.</small>}
          </div>
          <div className="admin-field">
            <label>Rôle *</label>
            <select name="role" value={form.role} onChange={handleChange}>
              {ROLES.map(r => <option key={r} value={r}>{r}</option>)}
            </select>
          </div>
          <div className="admin-field">
            <label>Date d'expiration du compte (optionnel)</label>
            <input name="dateExpiration" type="date"
              value={form.dateExpiration} onChange={handleChange} />
          </div>
          <div className="admin-field">
            <label>Modules accessibles</label>
            <div className="admin-modules-grid">
              {MODULES_DISPO.map(mod => (
                <label key={mod} className="admin-module-check">
                  <input type="checkbox"
                    checked={form.modules.includes(mod)}
                    onChange={() => toggleModule(mod)} />
                  {mod}
                </label>
              ))}
            </div>
          </div>
        </div>

        <div className="admin-modal-footer">
          <button onClick={onClose} className="admin-btn-annuler">Annuler</button>
          <button onClick={handleSave} disabled={chargement} className="admin-btn-primary">
            {chargement ? 'Enregistrement...' : (userAEditer ? 'Enregistrer' : 'Ajouter')}
          </button>
        </div>
      </div>
    </div>
  );
}

import React, { useState, useEffect } from 'react';
import { lireLogs } from '../../../services/adminService';

const ACTION_LABELS = {
  CREATE_USER:     { label: 'Création compte',    color: '#43A047' },
  UPDATE_USER:     { label: 'Modification',        color: '#1565C0' },
  CHANGE_ROLE:     { label: 'Changement de rôle', color: '#7E57C2' },
  ACTIVATE_USER:   { label: 'Activation',          color: '#00897B' },
  DEACTIVATE_USER: { label: 'Suspension',          color: '#F57F17' },
  DELETE_USER:     { label: 'Suppression',         color: '#E53935' },
  RESET_PASSWORD:  { label: 'Reset mot de passe', color: '#546E7A' },
};

export default function JournalAdmin() {
  const [logs,       setLogs]       = useState([]);
  const [chargement, setChargement] = useState(true);

  useEffect(() => {
    lireLogs()
      .then(setLogs)
      .catch(console.error)
      .finally(() => setChargement(false));
  }, []);

  if (chargement) return <div className="admin-loading">Chargement du journal...</div>;

  return (
    <div>
      <h2>Journal d'administration ({logs.length} entrées)</h2>
      <table className="admin-table">
        <thead>
          <tr>
            <th>Date</th>
            <th>Action</th>
            <th>Détails</th>
            <th>Résultat</th>
          </tr>
        </thead>
        <tbody>
          {logs.length === 0 && (
            <tr><td colSpan={4} style={{textAlign:'center',color:'#90A4AE'}}>
              Aucune action enregistrée
            </td></tr>
          )}
          {logs.map(log => {
            const info = ACTION_LABELS[log.action] || { label: log.action, color: '#999' };
            const date = log.timestamp?.toDate?.()
              ? log.timestamp.toDate().toLocaleString('fr-FR')
              : '—';
            return (
              <tr key={log.id}>
                <td style={{ fontSize:12, color:'#546E7A' }}>{date}</td>
                <td>
                  <span style={{ background: info.color + '22', color: info.color,
                    padding:'2px 10px', borderRadius:20, fontSize:12, fontWeight:600 }}>
                    {info.label}
                  </span>
                </td>
                <td style={{ fontSize:12 }}>
                  {JSON.stringify(log.details || {})}
                </td>
                <td style={{ color: log.resultat === 'SUCCESS' ? '#43A047' : '#E53935',
                  fontSize:12, fontWeight:600 }}>
                  {log.resultat}
                </td>
              </tr>
            );
          })}
        </tbody>
      </table>
    </div>
  );
}

Résultat attendu : email envoyé via Firebase Auth.

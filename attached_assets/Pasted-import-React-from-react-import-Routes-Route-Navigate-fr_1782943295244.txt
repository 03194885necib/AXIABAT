import React from "react";
import { Routes, Route, Navigate } from "react-router-dom";
import { AuthProvider } from "./context/AuthContext";
import ProtectedRoute from "./components/auth/ProtectedRoute";

// Pages Auth — nouvelles versions
import Login from "./pages/auth/Login";
import SignUp from "./pages/auth/SignUp";
import ForgotPassword from "./pages/auth/ForgotPassword";
import Unauthorized from "./pages/auth/Unauthorized";

// Page Admin — Gestion Utilisateurs
import UserManagement from "./pages/admin/UserManagement";

// Pages existantes AXIABAT — conservées intactes
import FirstPage from "./Components/FirstPage";
import FicheProjet from "./Components/FicheProjet";
import JournalCHantier from "./Components/JournalCHantier";
import Delais3 from "./Components/Delais3";
import BudgetDashboard from "./Components/BudgetDashboard";
import GlobalDash from "./Components/GlobalDash";
import Rapport from "./Components/Rapport";

function App() {
  return (
    <AuthProvider>
      <Routes>
        {/* Routes publiques */}
        <Route path="/" element={<Navigate to="/login" replace />} />
        <Route path="/login" element={<Login />} />
        <Route path="/signup" element={<SignUp />} />
        <Route path="/forgot-password" element={<ForgotPassword />} />
        <Route path="/unauthorized" element={<Unauthorized />} />

        {/* Routes protégées — tout utilisateur connecté */}
        <Route path="/FirstPage" element={
          <ProtectedRoute><FirstPage /></ProtectedRoute>
        } />
        <Route path="/FicheProjet" element={
          <ProtectedRoute><FicheProjet /></ProtectedRoute>
        } />
        <Route path="/JournalChantier" element={
          <ProtectedRoute><JournalCHantier /></ProtectedRoute>
        } />
        <Route path="/Delais" element={
          <ProtectedRoute><Delais3 /></ProtectedRoute>
        } />
        <Route path="/Budget" element={
          <ProtectedRoute><BudgetDashboard /></ProtectedRoute>
        } />
        <Route path="/Dashboard" element={
          <ProtectedRoute><GlobalDash /></ProtectedRoute>
        } />
        <Route path="/Rapport" element={
          <ProtectedRoute><Rapport /></ProtectedRoute>
        } />

        {/* Route admin uniquement */}
        <Route path="/admin/utilisateurs" element={
          <ProtectedRoute roles={["admin"]}>
            <UserManagement />
          </ProtectedRoute>
        } />

        {/* 404 */}
        <Route path="*" element={<Navigate to="/login" replace />} />
      </Routes>
    </AuthProvider>
  );
}

export default App;
